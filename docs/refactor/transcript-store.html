Composition root · 缓存跨 VC 生存的机制

// AppDelegate.applicationWillFinishLaunching
let registry = TranscriptRegistryStore()
let ctx = AppContext(
    sessionManager: …,
    transcriptRegistry: registry,   // ← app-scope
    …
)
mainWindowCoordinator = MainWindowCoordinator(
    appContext: ctx, parent: self)

Registry 由 AppDelegate 强持有，通过 AppContext read-only 引用向下传。窗口开关不影响它——MainWindowCoordinator 死掉时只 detach 自己的 window scope 资源，ctx.app.transcriptRegistry 里的 stores 继续活着。

切走 VC → VC deinit → 订阅解掉；registry 里的 store 保留 blocks + layouts。切回来新 VC 用 registry.store(for: id) 拿回同一个 store，store.blocks 非空 → 跳过 loader，直接 reloadData，layouts 全命中，秒开。

性能账目

  • Tail-first：SDK 反向读，首批就是最新
  • 反向配对在 SDK 内做完
  • SDK：off-main IO + decode + 配对
  • Phase 1 tail：全同步，一 tick 出屏
  • Phase 2 older：typeset off-main，主线程只 insertRows
  • 切走再回来：blocks + layouts 全命中 registry，无 SDK 读、无 BlockBuilder、无 typeset
  • heightOfRow 全走 Store cache

TranscriptRegistryStoreSTORE

@MainActor
final class TranscriptRegistryStore {

    private var stores: [String: TranscriptStore] = [:]

    func store(for id: String) -> TranscriptStore {
        if let s = stores[id] { return s }
        let s = TranscriptStore(id: id)
        stores[id] = s
        return s
    }

    /// SessionManager 归档 / 删除 session 时才调。关窗、切走都不调。
    func discard(_ id: String) {
        stores.removeValue(forKey: id)
    }
}

TranscriptStoreSTORE

@MainActor
final class TranscriptStore {

    enum BlockDelta {
        case tail([Block])     // 首批（最新）
        case older([Block])    // 后续更老页
    }

    let id: String
    private(set) var blocks: [Block] = []
    private(set) var layouts: [Block.ID: RowLayout] = [:]
    private(set) var layoutsWidth: CGFloat = 0
    let events = PassthroughSubject<BlockDelta, Never>()

    private var messages: [Message2] = []       // 私有工作面
    private var loader: Task<Void, Never>?
    private var isFirst = true

    init(id: String) { self.id = id }

    /// VC 在 viewWillAppear 里调；blocks 非空的话直接返回不重跑。
    func loadHistoryIfNeeded() {
        guard blocks.isEmpty, loader == nil else { return }
        loader = Task { [weak self] in
            guard let self else { return }
            let stream = AgentSDK.SessionHistory.load(id: id, order: .reverse)
            do {
                for try await batch in stream { self.ingest(batch) }
            } catch { appLog(.error, "TranscriptStore", "\(error)") }
        }
    }

    private func ingest(_ batch: [Message2]) {
        let newBlocks = BlockBuilder.build(from: batch, prior: messages)
        if isFirst {
            messages = batch
            blocks = newBlocks
            isFirst = false
            events.send(.tail(newBlocks))
        } else {
            messages = batch + messages
            blocks = newBlocks + blocks
            events.send(.older(newBlocks))
        }
    }

    /// heightOfRow / viewFor 走这里——miss 时同步 typeset + 写 cache。
    func layout(for block: Block, width: CGFloat) -> RowLayout {
        invalidateIfWidthChanged(width)
        if let l = layouts[block.id] { return l }
        let fresh = RowLayout.make(for: block, width: width)
        layouts[block.id] = fresh
        return fresh
    }

    /// Phase 2 后台 typeset 完批量回写。
    func writeLayouts(_ pairs: [(Block.ID, RowLayout)], width: CGFloat) {
        invalidateIfWidthChanged(width)
        for (id, l) in pairs { layouts[id] = l }
    }

    private func invalidateIfWidthChanged(_ w: CGFloat) {
        if w != layoutsWidth {
            layouts.removeAll(keepingCapacity: true)
            layoutsWidth = w
        }
    }
}

TranscriptViewControllerCONTROLLER

@MainActor
final class TranscriptViewController: NSViewController,
        NSTableViewDataSource, NSTableViewDelegate {

    private let store: TranscriptStore
    private var scrollView: NSScrollView!
    private var tableView: NSTableView!
    private var bag = Set<AnyCancellable>()

    /// Coordinator 从 registry 取 store 后注入。
    init(store: TranscriptStore) {
        self.store = store
        super.init(nibName: nil, bundle: nil)
    }

    override func loadView() { /* build scrollView + tableView; 不 super */ }

    override func viewDidLoad() {
        super.viewDidLoad()
        tableView.dataSource = self
        tableView.delegate   = self
        store.events
            .sink { [weak self] delta in self?.apply(delta) }
            .store(in: &bag)
        if !store.blocks.isEmpty {
            tableView.reloadData()                  // 缓存命中的 bootstrap
            tableView.scrollRowToVisible(store.blocks.count - 1)
        }
    }

    override func viewWillAppear() {
        super.viewWillAppear()
        store.loadHistoryIfNeeded()             // blocks 空才起 loader
    }

    func prepareForRemoval() {
        bag.removeAll()
        tableView.dataSource = nil
        tableView.delegate   = nil
        // 不动 store——它在 registry 里继续活着，下次挂载还用它
    }

    // ────────── Phase 1 sync / Phase 2 async ──────────

    private func apply(_ delta: TranscriptStore.BlockDelta) {
        let w = contentWidth
        switch delta {
        case .tail(let blocks):                    // Phase 1 · 同步链
            let pairs = blocks.map { ($0.id, RowLayout.make(for: $0, width: w)) }
            store.writeLayouts(pairs, width: w)
            let start = tableView.numberOfRows
            tableView.insertRows(at: IndexSet(integersIn: start..<start+blocks.count),
                                 withAnimation: [])
            tableView.scrollRowToVisible(tableView.numberOfRows - 1)

        case .older(let blocks):                   // Phase 2 · off-main typeset
            Task.detached(priority: .userInitiated) { [weak self] in
                let pairs = blocks.map { ($0.id, RowLayout.make(for: $0, width: w)) }
                await MainActor.run {
                    guard let self else { return }
                    self.store.writeLayouts(pairs, width: w)
                    self.tableView.insertRows(at: IndexSet(integersIn: 0..<blocks.count),
                                              withAnimation: [])
                }
            }
        }
    }

    // ────────── data source / delegate — 只读 store ──────────

    func numberOfRows(in tv: NSTableView) -> Int {
        store.blocks.count
    }

    func tableView(_ tv: NSTableView, heightOfRow row: Int) -> CGFloat {
        store.layout(for: store.blocks[row], width: contentWidth).height
    }

    func tableView(_ tv: NSTableView, viewFor col: NSTableColumn?, row: Int) -> NSView? {
        let b = store.blocks[row]
        let l = store.layout(for: b, width: contentWidth)
        return dequeueCell(for: b, layout: l, in: tv)
    }
}

关键决策一览

层数
四层：SDK → TranscriptRegistryStore → TranscriptStore → VC。VM 层删除——跨 VC 生存的东西按规范就是 Store 不是 VM。
缓存 scope
TranscriptRegistryStore 是 app-scope，由 AppDelegate 组合根 new，挂进 AppContext.transcriptRegistry。切窗 / 切走 / 切回来都不释放；只在 sidebar 归档 / 删除 session 时 discard(id)。
Session
零引用。VC 拿 TranscriptStore，Store 只带一个 String id。
history 数据源
AgentSDK.SessionHistory.load(id:order:) 静态 API，返回 AsyncThrowingStream<[Message2], Error>。
反向配对
SDK 内部。孤儿 tool_result buffer → 找到 tool_use 后正向 emit。文件顶未配对的作为最后一批 emit。
Batch 大小
SDK I/O buffer 决定，caller 不传参不控制。
Store 写 API
loadHistoryIfNeeded() 起消费；ingest(_:) 内部私有，跑 BlockBuilder.build 后同步 events.send(.tail | .older)。[Message2] 是 Store 私有工作面，公共 API 只见 Block。
Layout cache 归属
TranscriptStore.layouts: [Block.ID: RowLayout]。VC 不留本地 cache，heightOfRow / viewFor 全走 store.layout(for:width:)——miss 时 Store 同步 typeset + 写回。宽度变了整表清空。
Phase 1 · tail
主线程同步链：typeset → store.writeLayouts → insertRows → scrollToTail。首屏 100-200ms 主线程阻塞可接受（白屏更糟）。
Phase 2 · older
Task.detached off-main typeset → await MainActor.run { store.writeLayouts + insertRows }。主线程只吃 <5ms/批。
Off-main 用几处
两处：SDK 内 IO + decode + 配对；VC Phase 2 typeset。其他全主线程同步。
切回来命中
新 VC 从 registry 拿回同一 store。store.blocks 非空 → 跳过 loader；store.layouts 全命中（宽度没变）→ heightOfRow 秒回，无 typeset。
Delta 词汇
PassthroughSubject<BlockDelta, Never>——.tail([Block]) / .older([Block])。没有 append/update/remove，live 词汇之后再加。
取消 / 生命周期
VC prepareForRemoval：卸订阅 + 解绑 dataSource/delegate。不 cancel store.loader——loader 可能还在跑，让它跑完；下次挂载直接用结果。真正的淘汰是 SessionManager 归档时通过 registry.discard(id)。
live 演进接口
之后 live 是 TranscriptStore 的第二个 writer 入口（append/update/remove）+ 新加 BlockDelta case。现在不预留。
